**Goal:** 实现「AI 取名(名笺)」:输入姓氏+期望 → 免费 5 名+五维评分(音形义/五行/数理/笔画/性别,含古籍出处)→ 付费解锁深度报告(八字用神→五行补益矩阵/备选扩展/改名对比,¥19.9/¥29.9)→ 墨韵「名笺」呈现(收藏/分享),三版小程序同步,与姓名学页共存互打。

**Architecture:** 双引擎驱动:`BaziEngine`(排盘:四柱/五行分布/用神)为五行维度提供"缺啥补啥"的深层推理,`XingmingEngine`(五格/81数理/三才)为数理维度直接复用;新增取名用字库(五行/性别/风格/释义/古籍出处)+五维评分引擎+生成器;免费名单第 4、5 名只露综合分(服务端隐藏);深度报告按商品 `naming_report`(19.9)/`naming_adult`(29.9)走现有支付通道,会员免费;名笺收藏走本地存储并汇入收藏页"名"分类;三版前端以墨韵为准同步。

**Tech Stack:** FastAPI + sqlite(现有 dao 模式,`get_conn`)、现有 `BaziEngine`/`XingmingEngine`/支付通道(`/api/pay/create`+`/api/user/purchase`)、小程序原生(三版)、测试用 `scripts/test_*.py`(TestClient+临时 DB+AuthHandler 签发的假 token)。

"""取名用字库完整性测试:数量/去重/字段合法/出处可溯源/笔画可解析/无负面字"""
import os, sys
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
from src.engines.naming_chars import (CHAR_DB, CHAR_STROKES, ORIGINS, STYLES,
                                      NEGATIVE_WORDS, load_char_map, get_char_info, filter_chars)
from src.engines.xingming import STROKE_TABLE

ok = 0
def check(name, cond):
    global ok
    assert cond, f"FAIL: {name}"
    ok += 1; print(f"PASS: {name}")

WX = ("金", "木", "水", "火", "土")
GENDERS = ("男", "女", "通用")

# 1. 数量与去重
check("字库 ≥ 300 字", len(CHAR_DB) >= 300)
chars = [row[0] for row in CHAR_DB]
check("无重复字", len(set(chars)) == len(chars))

# 2. 字段合法
bad_wx = [c for c, w, g, ss, m in CHAR_DB if w not in WX]
check("五行合法", not bad_wx)
bad_g = [c for c, w, g, ss, m in CHAR_DB if g not in GENDERS]
check("性别合法", not bad_g)
bad_s = [c for c, w, g, ss, m in CHAR_DB if not ss or any(s not in STYLES for s in ss)]
check("风格合法且非空", not bad_s)
bad_m = [c for c, w, g, ss, m in CHAR_DB if not m or len(m) > 40]
check("释义非空且 ≤40 字", not bad_m)

# 3. 出处:只标字库内字
check("出处均可在字库溯源", set(ORIGINS.keys()) <= set(chars))
for c, (book, quote) in ORIGINS.items():
    assert book.startswith("《") and book.endswith("》") and quote, c
check("出处格式合法(书名带书名号+原句非空)", True)

# 4. 笔画可解析:每个字要么在 STROKE_TABLE 要么在 CHAR_STROKES
unresolved = [c for c in chars if c not in STROKE_TABLE and c not in CHAR_STROKES]
check("笔画全部可解析", not unresolved)

# 5. 负面字不进库
check("无负面联想字", not (set(chars) & set(NEGATIVE_WORDS)))

# 6. 工具函数
cmap = load_char_map()
check("load_char_map 覆盖全库", len(cmap) == len(chars))
check("get_char_info 命中", get_char_info("云") and get_char_info("云")["wuxing"] == "水")
check("get_char_info 未命中返回 None", get_char_info("X") is None)
f1 = filter_chars(gender="女", styles=["诗意"])
check("filter_chars 女+诗意 非空", len(f1) > 10)
check("filter_chars 性别过滤正确", all(get_char_info(c)["gender"] in ("女", "通用") for c in f1))
f2 = filter_chars(wuxing="水")
check("filter_chars 五行过滤", f2 and all(get_char_info(c)["wuxing"] == "水" for c in f2))

print(f"\nALL PASS ({ok})")

def load_char_map() -> dict:
    """字 → {char,wuxing,gender,styles,meaning} 映射(缓存)。"""
    if not _CACHE:
        for c, w, g, ss, m in CHAR_DB:
            _CACHE[c] = {"char": c, "wuxing": w, "gender": g, "styles": tuple(ss), "meaning": m}

def get_char_info(char: str) -> Optional[dict]:
    return load_char_map().get(char)


def filter_chars(gender: str = None, styles: list = None, wuxing: str = None) -> list:
    """按性别倾向/风格/五行过滤字库,返回字列表。"""
    out = []
    for c, w, g, ss, m in CHAR_DB:
        if gender and g not in ("通用", gender):
            continue
        if styles and not (set(ss) & set(styles)):
            continue
        if wuxing and w != wuxing:
            continue
        out.append(c)
    return out

"""AI 取名五维评分引擎(方案 §五)。

维度与权重:音形义 30% / 五行 25% / 数理 25% / 笔画 10% / 性别 10%。
- 数理维度直接复用 xingming.XingmingEngine(五格/81数理/三才);
- 五行维度由八字用神上下文驱动:缺啥补啥、过剩则泄;
- 各维度原始分按满分归一化到 0-100,综合分 = Σ 维度% × 权重(百分制)。
"""
import re
from typing import Optional

from src.engines.xingming import XingmingEngine, STROKE_TABLE
from src.engines.naming_chars import load_char_map, NEGATIVE_WORDS, CHAR_STROKES

WEIGHTS = {"音形义": 0.30, "五行": 0.25, "数理": 0.25, "笔画": 0.10, "性别": 0.10}
DIM_MAXES = {"音形义": 30, "五行": 25, "数理": 25, "笔画": 10, "性别": 10}
_ENGINE = XingmingEngine()
_YS_RE = re.compile(r"^([金木水火土])为用神.*（喜([金木水火土、]+)）")


def parse_yongshen(s: str):
    """解析排盘引擎用神文案:「X为用神(…)（喜A、B）」→ (用神元素, 喜元素列表)。"""
    if not s:
        return "", []
    m = _YS_RE.match(s.strip())
    if not m:
        return "", []
    return m.group(1), [c for c in m.group(2) if c in "金木水火土"]


def build_wuxing_ctx(bazi_result) -> dict:
    """从排盘结果构建五行上下文:use 用神/xi 喜神/counts 八字五行分布。
    用神解析失败时兜底:取八字最缺的五行(缺啥补啥)。"""
    counts = dict(bazi_result.wuxing or {})
    use, xi = parse_yongshen(getattr(bazi_result, "yongshen", "") or "")
    if not use:
        if counts:
            use = min(counts, key=lambda k: counts[k])
        xi = [use] if use else []
    elif use not in xi:
        xi.insert(0, use)
    return {"use": use, "xi": xi, "counts": counts}


def stroke_of(char: str) -> int:
    return STROKE_TABLE.get(char, CHAR_STROKES.get(char, 0))


def bu_yi_of(given: str, ctx: Optional[dict]) -> dict:
    """五行补益矩阵:该名对金木水火土的补益值(缺啥补啥,过剩则泄)。"""
    info = load_char_map()
    out = {}
    counts = (ctx or {}).get("counts", {})
    use = (ctx or {}).get("use", "")
    xi = (ctx or {}).get("xi", [])
    for c in given:
        w = (info.get(c) or {}).get("wuxing")
        if not w:
            continue
        if ctx:
            if w == use:
                v = 8
            elif w in xi:
                v = 6
            elif counts.get(w, 0) == 0:
                v = 6
            elif counts.get(w, 0) == 1:
                v = 4
            elif counts.get(w, 0) == 2:
                v = 2
            else:
                v = -3  # 过剩则泄
        else:
            v = 3
        out[w] = out.get(w, 0) + v
    return out


def _score_sound(given: str) -> tuple:
    """音形义:基础 24;叠字 -4;生僻(不在字库) -6;负面字 → (0, reject=True)。"""
    info = load_char_map()
    if any(c in NEGATIVE_WORDS for c in given):
        return 0, True
    s = 24
    if len(set(given)) < len(given):
        s -= 4
    if any(c not in info for c in given):
        s -= 6
    return max(0, min(30, s)), False


def _score_numerology(surname: str, given: str, gender: str) -> int:
    """数理:复用 XingmingEngine 五格/81数理/三才。"""
    try:
        res = _ENGINE.analyze(surname, given, gender)
    except Exception:
        return 10
    gs = list(res.analysis.values())
    big = sum(1 for g in gs if g.get("吉凶") == "大吉")
    ji = sum(1 for g in gs if g.get("吉凶") == "吉")
    n = 14 + 2 * big + ji
    if res.sancai_ji in ("吉", "大吉"):
        n += 3
    elif res.sancai_ji == "半吉":
        n += 1
    if res.analysis.get("人格", {}).get("吉凶") == "凶" or \
       res.analysis.get("总格", {}).get("吉凶") == "凶":
        n -= 8
    return max(0, min(25, n))


def _score_strokes(surname: str, given: str) -> int:
    """笔画:总笔画适中+单字均衡+无未知笔画。"""
    st = [stroke_of(c) for c in surname + given]
    p = 7
    if 0 in st:
        p -= 3
    if not (6 <= sum(st) <= 28):
        p -= 2
    if all(3 <= x <= 15 for x in st):
        p += 3
    return max(0, min(10, p))


def _score_gender(given: str, gender: str) -> int:
    info = load_char_map()
    g = 10
    for c in given:
        t = (info.get(c) or {}).get("gender", "通用")
        if t not in ("通用", gender):
            g -= 5
    return max(0, min(10, g))


def score_name(surname: str, given: str, gender: str, ctx: Optional[dict] = None) -> dict:
    """五维评分:返回 {dims: {维度: 0-100}, total: 0-100, bu_yi, reject}。"""
    dims = {}
    s, reject = _score_sound(given)
    dims["音形义"] = round(s / DIM_MAXES["音形义"] * 100)
    if ctx:
        dims["五行"] = round(min(25, sum(max(0, v) for v in bu_yi_of(given, ctx).values())) / 25 * 100)
    else:
        dims["五行"] = 48
    dims["数理"] = round(_score_numerology(surname, given, gender) / 25 * 100)
    dims["笔画"] = round(_score_strokes(surname, given) / 10 * 100)
    dims["性别"] = round(_score_gender(given, gender) / 10 * 100)
    total = round(sum(dims[k] * WEIGHTS[k] for k in dims))
    return {"dims": dims, "total": total, "bu_yi": bu_yi_of(given, ctx), "reject": reject}

"""AI 取名生成器:字库过滤(性别/风格/五行偏好)→ 枚举组合 → 五维评分 → 排序取前 N。

免费/付费边界在此引擎无感知(API 层裁剪);确定性由 seed 保证
(报告备选扩展依赖同请求同结果)。
"""
import random
from typing import Optional

from src.engines.naming_chars import filter_chars, get_char_info, ORIGINS
from src.engines.naming_score import score_name
from src.engines.xingming import XingmingEngine

_POOL_TOP = 80   # 按五行偏好取字池上限
_EXTRA_WEIGHT = {"音形义": 0, "五行": 1, "数理": 1, "笔画": 0, "性别": 0}


def _resolve_seed(surname: str, gender: str, birth_date: str) -> int:
    return abs(hash(f"{surname}|{gender}|{birth_date}")) % (2 ** 31)


def _prefer_pool(pool: list, ctx: Optional[dict]) -> list:
    """用神/喜神/缺门五行对应的字优先(排序靠前,保持池内确定性)。"""
    if not ctx:
        return pool
    use = ctx.get("use", "")
    xi = ctx.get("xi", [])
    counts = ctx.get("counts", {})
    weak = [w for w in "金木水火土" if counts.get(w, 0) < 2]

    def key(c):
        w = (get_char_info(c) or {}).get("wuxing", "")
        score = 0
        if w == use:
            score += 3
        elif w in xi:
            score += 2
        elif w in weak:
            score += 1
        return -score

    return sorted(pool, key=key)


def _build_comment(given: str, sc: dict, ctx: Optional[dict]) -> str:
    """一句人话点评:出处优先;无出处用释义组合;补益人话中性措辞(合规)。"""
    info = get_char_info
    parts = []
    origin_hit = None
    for c in given:
        if c in ORIGINS and not origin_hit:
            origin_hit = ORIGINS[c]
    if origin_hit:
        book, quote = origin_hit
        m0 = (info(given[0]) or {}).get("meaning", "")
        parts.append(f"出自{book}「{quote}」，{m0}")
    else:
        ms = [((info(c) or {}).get("meaning") or "").rstrip("，。") for c in given]
        parts.append("，".join(m for m in ms if m))
    if ctx and sc.get("bu_yi"):
        pos = {k: v for k, v in sc["bu_yi"].items() if v > 0}
        if pos:
            buf = "、".join(f"{k}+{v}" for k, v in pos.items())
            counts = ctx.get("counts", {})
            strong = max(counts, key=counts.get) if counts else ""
            weak = "、".join(w for w in "金木水火土" if counts.get(w, 0) <= 1)[:2]
            if not weak and counts:
                weak = min(counts, key=counts.get)
            parts.append(f"八字{strong}偏旺、{weak}偏弱，此名补{buf}，与用神相合")
    return "，".join(parts) + "。"


def _score_candidate(surname: str, given: str, gender: str, ctx: Optional[dict]) -> dict:
    sc = score_name(surname, given, gender, ctx)
    if sc.get("reject"):
        return None
    origin = ""
    for c in given:
        if c in ORIGINS:
            book, quote = ORIGINS[c]
            origin = f"{book}"
            break
    return {
        "name": given,
        "total": sc["total"],
        "dims": sc["dims"],
        "bu_yi": sc["bu_yi"],
        "comment": _build_comment(given, sc, ctx),
        "origin": origin,
        "wuxing_dim": sc["dims"]["五行"],
        "shuli_dim": sc["dims"]["数理"],
    }


def _generate_pool(gender: str, style: str, ctx: Optional[dict]) -> list:
    pool = filter_chars(gender=gender, styles=[style] if style else None)
    pool = _prefer_pool(pool, ctx)
    return pool[:_POOL_TOP]


def _generate_all(surname: str, gender: str, style: str, ctx: Optional[dict],
                  seed: int, limit: int) -> list:
    pool = _generate_pool(gender, style, ctx)
    rng = random.Random(seed)
    cands = []
    for a in pool:
        cands.append(a)              # 单字名
        for b in pool:
            cands.append(a + b)      # 双字名
    scored = []
    for given in cands:
        if len(set(given)) < len(given):
            continue                 # 叠字剔除
        item = _score_candidate(surname, given, gender, ctx)
        if item is None:
            continue
        item["_tie"] = rng.random()
        scored.append(item)
    scored.sort(key=lambda x: (-x["total"], -x["wuxing_dim"], -x["shuli_dim"], x["_tie"]))
    return scored[:limit]


def _pack(items: list, start_rank: int, unlocked: bool) -> list:
    out = []
    for i, it in enumerate(items):
        rank = start_rank + i
        if unlocked:
            out.append({
                "rank": rank, "name": it["name"], "total": it["total"],
                "dims": it["dims"], "bu_yi": it["bu_yi"],
                "comment": it["comment"], "origin": it["origin"],
            })
        else:
            out.append({"rank": rank, "name": it["name"], "total": it["total"], "unlocked": False})
    return out


def generate_names(surname: str, gender: str, style: str = "", ctx: Optional[dict] = None,
                   count: int = 5, seed: int = 0) -> list:
    """免费主榜 5 名(前 3 名完整,后 2 名压缩形态——压缩在 API 层做,这里全量)。"""
    items = _generate_all(surname, gender, style, ctx, seed, count)
    return _pack(items, 1, unlocked=True)


def generate_extended(surname: str, gender: str, style: str = "", ctx: Optional[dict] = None,
                      count: int = 15, seed: int = 0) -> list:
    """备选扩展 15 名(付费,仅带 name/total/origin,防复制)。"""
    items = _generate_all(surname, gender, style, ctx, seed, count + 5)[5:]
    return [{"rank": i + 6, "name": it["name"], "total": it["total"],
             "origin": it["origin"], "comment": it["comment"]}
            for i, it in enumerate(items)]


def generate_adult_diagnosis(surname: str, given: str, ctx: Optional[dict] = None) -> dict:
    """成人改名免费诊断:数理凶格/三才问题/五行冲突 + 中性建议(不做改运承诺)。"""
    issues = []
    res = XingmingEngine().analyze(surname, given, "男")
    for k, v in res.analysis.items():
        if v.get("吉凶") == "凶":
            issues.append(f"{k}数理{v.get('数字')}为凶")
    if res.sancai_ji in ("凶", "半吉"):
        issues.append(f"三才配置{res.sancai_ji}（{res.sancai}）")
    if ctx:
        counts = ctx.get("counts", {})
        if counts:
            strong = max(counts, key=counts.get)
            for c in given:
                w = (get_char_info(c) or {}).get("wuxing", "")
                if w and w == strong and counts.get(w, 0) >= 3:
                    issues.append(f"「{c}」属{w}，与八字中偏旺的{w}相重")
    suggestion = "综合来看可以保留现名；如介意上述数理或五行分布，可考虑调整用字。" \
                 if issues else "现名数理与三才配置均平稳，可以保留。"
    return {"issues": issues, "suggestion": suggestion}

"""AI 取名存储:生成记录(naming_records)+ 每日免费限频(naming_usage)。

安全:记录归属 user_id 落库;API 层只允许读取自己的记录(方案 §八 防爬防刷对齐)。
"""
import json
import time


class NamingRecordDAO:
    def __init__(self, conn):
        self.conn = conn
        self.conn.execute("""CREATE TABLE IF NOT EXISTS naming_records (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id TEXT NOT NULL,
            request_json TEXT NOT NULL,
            results_json TEXT NOT NULL,
            created_at REAL
        )""")
        self.conn.execute("CREATE INDEX IF NOT EXISTS idx_naming_records_user ON naming_records(user_id)")
        self.conn.execute("""CREATE TABLE IF NOT EXISTS naming_usage (
            user_id TEXT NOT NULL,
            date TEXT NOT NULL,
            count INTEGER DEFAULT 0,
            PRIMARY KEY (user_id, date)
        )""")
        self.conn.commit()

    def create_record(self, user_id: str, request_json: str, results_json: str) -> int:
        cur = self.conn.execute(
            "INSERT INTO naming_records (user_id, request_json, results_json, created_at)"
            " VALUES (?,?,?,?)",
            (user_id, request_json, results_json, time.time()))
        self.conn.commit()
        return cur.lastrowid

    def get_record(self, record_id: int):
        row = self.conn.execute(
            "SELECT id, user_id, request_json, results_json, created_at"
            " FROM naming_records WHERE id=?", (record_id,)).fetchone()
        if not row:
            return None
        cols = [d[0] for d in self.conn.execute("SELECT * FROM naming_records").description]
        return dict(zip(cols, row))

    def list_records(self, user_id: str, limit: int = 20) -> list:
        rows = self.conn.execute(
            "SELECT id, user_id, request_json, results_json, created_at"
            " FROM naming_records WHERE user_id=? ORDER BY id DESC LIMIT ?",
            (user_id, limit)).fetchall()
        cols = [d[0] for d in self.conn.execute("SELECT * FROM naming_records").description]
        return [dict(zip(cols, r)) for r in rows]

    def daily_generations(self, user_id: str, date: str) -> int:
        row = self.conn.execute(
            "SELECT count FROM naming_usage WHERE user_id=? AND date=?",
            (user_id, date)).fetchone()
        return row[0] if row else 0

    def inc_generation(self, user_id: str, date: str) -> int:
        self.conn.execute(
            "INSERT INTO naming_usage (user_id, date, count) VALUES (?,?,1)"
            " ON CONFLICT(user_id, date) DO UPDATE SET count=count+1",
            (user_id, date))
        self.conn.commit()
        return self.daily_generations(user_id, date)

合规红线(方案 §四/§十):

import json

from datetime import datetime, timezone, timedelta
from typing import Optional

from fastapi.responses import JSONResponse
from pydantic import BaseModel, field_validator

def _today() -> str:
    return datetime.now(timezone(timedelta(hours=8))).strftime("%Y-%m-%d")


class NamingGenerateRequest(BaseModel):
    surname: str
    gender: str = "女"
    birth_date: str
    birth_hour: Optional[int] = None
    style: str = ""
    scene: str = "baby"
    current_name: Optional[str] = None

    @field_validator("gender")
    def _gender(cls, v):
        if v not in ("男", "女"):
            raise ValueError("性别必须为男或女")
        return v

    @field_validator("birth_date")
    def _date(cls, v):
        try:
            datetime.strptime(v, "%Y-%m-%d")
        except ValueError:
            raise ValueError(f"非法出生日期: {v}")
        return v

    @field_validator("birth_hour")
    def _hour(cls, v):
        if v is not None and not (0 <= v <= 23):
            raise ValueError(f"非法时辰: {v}")
        return v

    def _style(cls, v):
        if v and v not in STYLES:
            raise ValueError(f"非法风格: {v}（可选 {STYLES}）")
        return v

    @field_validator("scene")
    def _scene(cls, v):
        if v not in ("baby", "adult"):
            raise ValueError("scene 必须为 baby 或 adult")
        return v

@router.get("/quota")

@router.post("/generate")
def generate(req: NamingGenerateRequest, uid: str = Depends(require_user)):
    # 校验先于限频:限频只挡"合法请求",参数错误永远 400(pydantic 已拦格式类)
    if req.scene == "adult" and not (req.current_name or "").strip():
        raise HTTPException(status_code=400, detail="成人改名需提供现名")
    dao = _dao()
    if dao.daily_generations(uid, _today()) >= FREE_DAILY_LIMIT:
        raise HTTPException(status_code=429, detail="今日免费生成已达 3 次，明日再来",
                            headers={"X-RateLimit-Remaining": "0"})

    y, m, d = (int(x) for x in req.birth_date.split("-"))
    hour = req.birth_hour
    hour_fallback = hour is None
    if hour is None:
        hour = 0  # 子时兜底

    bazi = BaziEngine().calculate(y, m, d, hour, 0, "北京", req.gender)
    ctx = build_wuxing_ctx(bazi)
    seed = _resolve_seed(req.surname, req.gender, req.birth_date)

    full_names = generate_names(req.surname, req.gender, req.style, ctx, count=5, seed=seed)
    # 免费边界:第 4、5 名只露综合分(服务端裁剪)
    names = []
    for n in full_names:
        if n["rank"] <= 3:
            names.append(n)
        else:
            names.append({"rank": n["rank"], "name": n["name"],
                          "total": n["total"], "unlocked": False})

        "surname": req.surname, "gender": req.gender, "scene": req.scene,
        "hour_fallback": hour_fallback,
        "bazi": {
            "pillars": bazi.bazi,
            "day_master": bazi.day_master,
            "wuxing": bazi.wuxing,
            "yongshen": bazi.yongshen,
        },
        "names": names,
        "footer": FOOTER,
    }
    if req.scene == "adult":
        given = (req.current_name or "").strip()[len(req.surname):] or (req.current_name or "").strip()
        generation["adult_diagnosis"] = generate_adult_diagnosis(req.surname, given, ctx)

    record_id = dao.create_record(uid, json.dumps(req.model_dump(), ensure_ascii=False),
                                  json.dumps({"generation": generation}, ensure_ascii=False))
    used = dao.inc_generation(uid, _today())
    return {"record_id": record_id, "generation": generation,

def _unlocked(uid: str, product_id: str) -> bool:
    """会员或已购对应商品 → 解锁(方案 §七:会员深度报告全免费)。"""
    from src.config import is_experience_mode
    if is_experience_mode():
        return True
    if _member_dao:
        try:
            mem = _member_dao.get_membership(uid)
            if mem and mem.get("plan") != "free":
                return True
        except Exception as e:
            logger.warning("会员判定失败: %s", e)
        try:
            if _member_dao.get_user_purchase(uid, product_id):
                return True
        except Exception as e:
            logger.warning("购买判定失败: %s", e)
    return False


def _build_reasoning(name_item: dict, bazi: dict, ctx: dict) -> str:
    """用神推理原文(付费核心):引擎输出原文 + 补益人话(与引擎表述严格一致,禁编造)。"""
    pos = {k: v for k, v in name_item.get("bu_yi", {}).items() if v > 0}
    neg = {k: v for k, v in name_item.get("bu_yi", {}).items() if v < 0}
    counts = bazi.get("wuxing", {})
    parts = [f"排盘引擎用神判定:{bazi.get('yongshen','')}。",
             f"八字五行分布:金{counts.get('金',0)}、木{counts.get('木',0)}、水{counts.get('水',0)}、火{counts.get('火',0)}、土{counts.get('土',0)}。"]
    if pos:
        buf = "、".join(f"{k}+{v}" for k, v in pos.items())
        parts.append(f"此名补{buf}")
    if neg:
        nbuf = "、".join(f"{k}{v}" for k, v in neg.items())
        parts.append(f"对{nbuf}，五行偏旺者宜泄不宜再补")
    parts.append("以上为传统文化视角的五行平衡参考，不构成任何科学或命运承诺。")
    return "。".join(parts)
